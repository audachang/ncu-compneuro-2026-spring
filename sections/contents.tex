\section{Weekly Content Details}

\subsection*{Module 1: Experimental Programming \& Data Science}
\noindent\textbf{Focus}: \textit{From Hypothesis to Data—Mastering the Python ecosystem to build robust behavioral experiments, ensure reproducibility, and visualize scientific results.}

\vspace{0.3cm}

\noindent\textbf{Week 1: Orientation, Python Environment Setup \& Basic Libraries}
\begin{itemize}
    \item \textbf{Date}: 2026/02/26
    \item \textbf{Topic}: \textbf{Setting the Stage for Scientific Computing}
    \item \textbf{Course Content}:
    \begin{itemize}
        \item \textbf{The Stack}: VS Code, Anaconda/Miniconda, and managing Virtual Environments (\texttt{conda}/\texttt{venv}).
        \item \textbf{Python Refresher}: Mutable vs. Immutable types, List Comprehensions, and Modular design.
        \item \textbf{Scientific Libs}: Introduction to \textbf{NumPy} for high-performance array manipulation (essential for coordinate systems, timing, and randomization).
    \end{itemize}
    \item \textbf{Teaching Plan}:
    \begin{itemize}
        \item \textbf{Lecture}: The "Reproducibility Crisis" and how code provides the solution.
        \item \textbf{Lab Activity (Install-Fest)}: Ensure every student has a working local environment.
        \item \textbf{Homework}: "Hello Data"—Write a script that generates a synthetic dataset of reaction times using NumPy.
    \end{itemize}
\end{itemize}

\noindent\textbf{Week 2: PsychoPy Coder \& Stimulus Presentation}
\begin{itemize}
    \item \textbf{Date}: 2026/03/05
    \item \textbf{Topic}: \textbf{Drawing to the Screen (Beyond the GUI)}
    \item \textbf{Course Content}:
    \begin{itemize}
        \item \textbf{The Window}: \texttt{visual.Window}, units (\texttt{norm}, \texttt{pix}, \texttt{deg}), and frame buffering (\texttt{win.flip()}).
        \item \textbf{Stimuli}: Programmatically creating \texttt{TextStim}, \texttt{ImageStim}, and \texttt{GratingStim} (Gabor patches).
        \item \textbf{Timing}: Understanding refresh rates (Hz), frame-based timing, and precision.
    \end{itemize}
    \item \textbf{Teaching Plan}:
    \begin{itemize}
        \item \textbf{Live Coding}: Build a script from scratch that displays a fixation cross followed by a drifting Gabor patch.
        \item \textbf{Lab}: Modify the script to change the orientation of the grating dynamically based on a frame counter.
    \end{itemize}
\end{itemize}

\noindent\textbf{Week 3: Interaction \& The Event Loop}
\begin{itemize}
    \item \textbf{Date}: 2026/03/12
    \item \textbf{Topic}: \textbf{The Game Loop of Research}
    \item \textbf{Course Content}:
    \begin{itemize}
        \item \textbf{Input}: Polling (\texttt{defaultKeyboard.getKeys()}) vs. Blocking (\texttt{event.waitKeys()}).
        \item \textbf{Architecture}: Implementing the Trial $\rightarrow$ Block $\rightarrow$ Experiment hierarchy.
        \item \textbf{Data Logging}: Using \texttt{pandas} or PsychoPy’s \texttt{ExperimentHandler} to save trial-by-trial data to CSV.
    \end{itemize}
    \item \textbf{Teaching Plan}:
    \begin{itemize}
        \item \textbf{Lab Activity (The Stroop Task)}: Build a complete interactive task where participants identify font colors of conflicting words.
        \item \textbf{Crucial Step}: Students must run the task on themselves to generate real data for Week 5.
    \end{itemize}
\end{itemize}

\noindent\textbf{Week 4: PsychoPy Builder, Online Paradigms \& Adaptive Design}
\begin{itemize}
    \item \textbf{Date}: 2026/03/19
    \item \textbf{Topic}: \textbf{Rapid Prototyping \& Web Deployment}
    \item \textbf{Course Content}:
    \begin{itemize}
        \item \textbf{Builder GUI}: Routines, Loops, and injecting Custom Code components.
        \item \textbf{Pavlovia}: Converting Python experiments to PsychoJS for online data collection.
        \item \textbf{Adaptive Design}: Introduction to \textbf{Staircase procedures} (e.g., Up/Down method) for finding sensory thresholds.
    \end{itemize}
    \item \textbf{Teaching Plan}:
    \begin{itemize}
        \item \textbf{Demo}: Recreate the Week 3 manual code using the Builder GUI in 15 minutes.
        \item \textbf{Workshop}: Push a simple experiment to Pavlovia and collect data from a peer.
    \end{itemize}
\end{itemize}

\noindent\textbf{Week 5: PsychoPy Experiment Design}
\begin{itemize}
    \item \textbf{Date}: 2026/03/26
    \item \textbf{Topic}: \textbf{From Raw Logs to Insights}
    \item \textbf{Course Content}:
    \begin{itemize}
        \item \textbf{Data Cleaning}: Using \textbf{Pandas} to filter outliers, group data, and handle missing values.
        \item \textbf{Visualization}: \textbf{Seaborn} and \textbf{Matplotlib} for Violin plots, Scatter plots, and Error bars.
        \item \textbf{Stats}: T-tests and correlations using \texttt{scipy.stats}.
    \end{itemize}
    \item \textbf{Teaching Plan}:
    \begin{itemize}
        \item \textbf{Data Dive}: Analyze the \textbf{Stroop Task data} generated in Week 3.
        \item \textbf{Deliverable}: A script that produces a publication-quality figure showing the "Stroop Effect" (RT difference).
    \end{itemize}
\end{itemize}

\noindent\textbf{Week 6: PsychoPy Builder, Stroop Experiment \& Pavlovia Deployment}
\begin{itemize}
    \item \textbf{Date}: 2026/04/02
    \item \textbf{Topic}: \textbf{Rapid Prototyping with Builder \& Online Deployment}
    \item \textbf{Course Content}:
    \begin{itemize}
        \item \textbf{Builder GUI}: Routines, Loops, Excel condition files, and injecting Custom Code components.
        \item \textbf{Stroop Experiment}: Building a complete Stroop task in Builder—stimulus file design, keyboard responses, automatic scoring (\texttt{correct\_key}), and pivot-table analysis.
        \item \textbf{Code Components}: Independently randomizing words and colors so any word can appear in any ink color.
        \item \textbf{Pavlovia}: Converting the experiment to PsychoJS for online data collection.
    \end{itemize}
    \item \textbf{Teaching Plan}:
    \begin{itemize}
        \item \textbf{Hands-on Workshop}: Build the Stroop experiment step-by-step in Builder and deploy to Pavlovia.
        \item \textbf{Data Analysis}: Collect pilot data and summarize results using pivot tables (RT × congruency).
    \end{itemize}
\end{itemize}

\noindent\textbf{Week 7: NumPy, Data Manipulation \& Visualization}
\begin{itemize}
    \item \textbf{Date}: 2026/04/09
    \item \textbf{Topic}: \textbf{From Raw Data to Publication-Quality Figures}
    \item \textbf{Course Content}:
    \begin{itemize}
        \item \textbf{NumPy}: Array creation, indexing, broadcasting, vectorized operations, and boolean masking.
        \item \textbf{Data Processing}: Reaction-time normalization (z-scores), outlier filtering, and batch processing with NumPy.
        \item \textbf{Matplotlib}: Line plots, scatter plots, bar charts with error bars, histograms, multi-panel figures (subplots), and annotations.
    \end{itemize}
    \item \textbf{Teaching Plan}:
    \begin{itemize}
        \item \textbf{Data Dive}: Analyze \textbf{Stroop task data} from Week 6—filter outliers with NumPy, compute condition means, and produce publication-quality figures showing the Stroop effect.
    \end{itemize}
\end{itemize}

\noindent\textbf{Week 8: Midterm Project Presentation}
\begin{itemize}
    \item \textbf{Date}: 2026/04/16
    \item \textbf{Activity}: \textbf{Seminar \& Demo}
    \item \textbf{Requirements}:
    \begin{enumerate}
        \item A fully functional Python/PsychoPy experiment.
        \item Automatic data logging.
        \item A visualization of pilot data.
        \item A clean GitHub repository link.
    \end{enumerate}
\end{itemize}

\subsection*{Module 2: Vibe/Agentic Coding \& Data Applications}

\noindent\textbf{Focus}: \textit{From describing intent to shipping applications—using AI as a coding partner to build, deploy, and narrate data-driven web apps grounded in real-world open data.}

\vspace{0.3cm}

\noindent\textbf{Week 9: Vibe/Agentic Coding Fundamentals}
\begin{itemize}
    \item \textbf{Date}: 2026/04/23
    \item \textbf{Topic}: \textbf{From Writing Code to Directing AI}
    \item \textbf{Course Content}:
    \begin{itemize}
        \item \textbf{The Spectrum}: Copilot autocomplete $\rightarrow$ Chat collaboration $\rightarrow$ Agentic execution.
        \item \textbf{Vibe Coding Philosophy}: Describe intent, let AI scaffold, human verifies; comparing ChatGPT/Claude web, Cursor IDE, and Claude Code CLI.
        \item \textbf{Setup}: Installing Node.js, Claude Code, API key management, first dialogue.
    \end{itemize}
    \item \textbf{Teaching Plan}:
    \begin{itemize}
        \item \textbf{Live Demo}: Have Claude Code read and improve a pandas notebook by prompt only.
        \item \textbf{Lab}: Rewrite Week 7's Stroop analysis script using Claude Code; compare original vs. AI-assisted versions.
        \item \textbf{Homework}: One-page reflection on where AI outperformed and where human judgment was essential.
    \end{itemize}
\end{itemize}

\noindent\textbf{Week 10: Multi-Step Task Delegation, Git \& GitHub Actions}
\begin{itemize}
    \item \textbf{Date}: 2026/04/30
    \item \textbf{Topic}: \textbf{Version Control in the Agentic Era}
    \item \textbf{Course Content}:
    \begin{itemize}
        \item \textbf{Prompt Design}: Plan--execute--verify pattern; \texttt{CLAUDE.md} and project context files.
        \item \textbf{Git Essentials}: Repo, commit, branch, push, pull via GitHub Desktop (GUI) and Claude Code (CLI).
        \item \textbf{GitHub Actions}: Automated checks on push (linting, Streamlit build verification).
    \end{itemize}
    \item \textbf{Teaching Plan}:
    \begin{itemize}
        \item \textbf{Demo}: Delegate an end-to-end pipeline (read CSV $\rightarrow$ clean $\rightarrow$ plot $\rightarrow$ commit $\rightarrow$ push) to Claude Code.
        \item \textbf{Lab}: Create personal project repo on GitHub; configure a minimal GitHub Action.
        \item \textbf{Homework}: Use AI to draft \texttt{README.md} and \texttt{CLAUDE.md} defining project goals and coding norms.
    \end{itemize}
\end{itemize}

\noindent\textbf{Week 11: Streamlit Fundamentals \& Cloud Deployment}
\begin{itemize}
    \item \textbf{Date}: 2026/05/07
    \item \textbf{Topic}: \textbf{From Script to Web App in Pure Python}
    \item \textbf{Course Content}:
    \begin{itemize}
        \item \textbf{Core API}: \texttt{st.write}, \texttt{st.dataframe}, widgets (\texttt{slider}, \texttt{selectbox}, \texttt{file\_uploader}), layout (\texttt{columns}, \texttt{tabs}).
        \item \textbf{State \& Caching}: \texttt{st.session\_state} and \texttt{@st.cache\_data}.
        \item \textbf{Deployment}: Streamlit Community Cloud with GitHub integration.
    \end{itemize}
    \item \textbf{Teaching Plan}:
    \begin{itemize}
        \item \textbf{Live Coding}: Vibe-code a Stroop analysis script into an interactive Streamlit app in 30 minutes.
        \item \textbf{Lab}: Convert a personal pandas/matplotlib analysis into a deployed Streamlit app.
        \item \textbf{Homework}: Submit deployed app URL.
    \end{itemize}
\end{itemize}

\noindent\textbf{Week 12: Taiwan Open Data \& Real-World Cleaning}
\begin{itemize}
    \item \textbf{Date}: 2026/05/12
    \item \textbf{Topic}: \textbf{APIs, Messy Data, and How AI Accelerates Cleaning}
    \item \textbf{Course Content}:
    \begin{itemize}
        \item \textbf{Data Sources}: data.gov.tw, TDX transport API, MOHW statistics, weather/air-quality APIs.
        \item \textbf{HTTP/REST}: GET/POST, query parameters, JSON responses; \texttt{requests} with \texttt{pandas.read\_json}/\texttt{read\_csv}.
        \item \textbf{Cleaning}: Chinese encoding, date parsing, missing values, deduplication, dtype coercion.
    \end{itemize}
    \item \textbf{Teaching Plan}:
    \begin{itemize}
        \item \textbf{Demo}: Claude Code analyzes a ``dirty'' air-quality dataset and proposes a cleaning pipeline with justifications.
        \item \textbf{Lab}: Each group selects a behaviorally/socially relevant dataset; completes Download $\rightarrow$ Clean $\rightarrow$ EDA notebook.
        \item \textbf{Homework}: Integrate the dataset into the Week 11 Streamlit app.
    \end{itemize}
\end{itemize}

\noindent\textbf{Week 13: Interactive Visualization \& Data Storytelling}
\begin{itemize}
    \item \textbf{Date}: 2026/05/14
    \item \textbf{Topic}: \textbf{Charts That Tell a Story}
    \item \textbf{Course Content}:
    \begin{itemize}
        \item \textbf{Storytelling Structure}: Hook $\rightarrow$ Context $\rightarrow$ Conflict $\rightarrow$ Insight $\rightarrow$ Conclusion.
        \item \textbf{Tools}: \texttt{plotly.express}, Altair, \texttt{pydeck} for geospatial data.
        \item \textbf{Design Principles}: Chart selection hierarchy, colorblind-safe palettes, misleading-axis anti-patterns.
    \end{itemize}
    \item \textbf{Teaching Plan}:
    \begin{itemize}
        \item \textbf{Comparison Demo}: Same data rendered as academic-style static figure vs. public-facing interactive chart.
        \item \textbf{Lab}: Add $\geq 3$ interactive elements (filters, tooltips, maps) and a narrative flow to the Streamlit app.
        \item \textbf{Homework}: 3-minute demo video, practicing storytelling delivery.
    \end{itemize}
\end{itemize}

\noindent\textbf{Week 14: Anthropic SDK \& Responsible AI Integration}
\begin{itemize}
    \item \textbf{Date}: 2026/05/21
    \item \textbf{Topic}: \textbf{Adding AI Features to Your App}
    \item \textbf{Course Content}:
    \begin{itemize}
        \item \textbf{Anthropic API}: messages endpoint, system prompts, roles, streaming.
        \item \textbf{Integration Patterns}: data summarization, Q\&A over data, natural-language-to-pandas, classification/sentiment.
        \item \textbf{Responsible AI}: hallucinations, privacy, de-identification, cost control, transparency (labeling AI output); alignment with EU AI Act and Taiwan's AI Basic Act draft.
    \end{itemize}
    \item \textbf{Teaching Plan}:
    \begin{itemize}
        \item \textbf{Demo}: Add an ``Ask this data'' dialog to the Streamlit app—natural language in, answer or chart out.
        \item \textbf{Lab}: Each group integrates one AI feature (summary / chat / auto-report) with \texttt{.env}/Streamlit secrets for key management.
        \item \textbf{Homework}: Add responsible-AI disclosure and API-cost estimate notes to the app.
    \end{itemize}
\end{itemize}

\begin{itemize}

    \end{itemize}
\end{itemize}
